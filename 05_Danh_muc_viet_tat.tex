\chapter*{DANH MỤC TỪ VIẾT TẮT}
\markboth{DANH MỤC TỪ VIẾT TẮT}{}
\begin{longtable}{l p{6cm} p{7cm}}
	\hline
  \textbf{ABAC} & Attribute-Based Access Control & Kiểm soát truy cập dựa trên thuộc tính\\
  \textbf{ACL} & Access Control List & Danh sách kiểm soát truy cập\\
  \textbf{API} & Application Programming Interface & Giao diện lập trình ứng dụng\\
  \textbf{BSON} & Binary JSON & Định dạng JSON nhị phân dùng trong MongoDB\\
  \textbf{CI/CD} & Continuous Integration / Continuous Deployment & Tích hợp và triển khai liên tục\\
  \textbf{COD} & Cash on Delivery & Thanh toán khi nhận hàng\\
  \textbf{CORS} & Cross-Origin Resource Sharing & Chia sẻ tài nguyên giữa các nguồn gốc khác nhau\\
  \textbf{CRUD} & Create, Read, Update, Delete & Tạo, đọc, cập nhật, xóa\\
  \textbf{CSDL} & & Cơ sở dữ liệu\\
  \textbf{CSS} & Cascading Style Sheets & Ngôn ngữ biểu diễn (style language) giúp định dạng và làm đẹp cho các trang web được viết bằng HTML hoặc XML\\
  \textbf{DOM} & Document Object Model & Mô hình đối tượng tài liệu\\
  \textbf{FSM} & Finite State Machine & Máy trạng thái hữu hạn\\
  \textbf{HMAC} & Hash-based Message Authentication Code & Mã xác thực thông điệp dựa trên hàm băm\\
  \textbf{HTTP} & HyperText Transfer Protocol & Giao thức truyền tải siêu văn bản\\
  \textbf{HTTPS} & HyperText Transfer Protocol Secure & Giao thức truyền tải siêu văn bản bảo mật\\
  \textbf{IDE} & Integrated Development Environment & Môi trường phát triển tích hợp\\
  \textbf{JSON} & JavaScript Object Notation & Định dạng dữ liệu dùng để lưu trữ và trao đổi dữ liệu\\
  \textbf{JWT} & JSON Web Token & Chuẩn xác thực dựa trên token\\
  \textbf{LTS} & Long Term Support & Phiên bản hỗ trợ dài hạn\\
  \textbf{MERN} & MongoDB, Express, React, Node.js & Bộ công nghệ phát triển web full stack\\
  \textbf{MVC} & Model - View - Controller & Mô hình kiến trúc phần mềm\\
  \textbf{NoSQL} & Not only SQL & Hệ quản trị cơ sở dữ liệu phi quan hệ\\
  \textbf{ODM} & Object Document Mapper & Ánh xạ đối tượng tài liệu (tương đương ORM cho NoSQL)\\
  \textbf{ORM} & Object Relational Mapping & Ánh xạ đối tượng quan hệ\\
  \textbf{RBAC} & Role-Based Access Control & Kiểm soát truy cập dựa trên vai trò\\
  \textbf{REST} & Representational State Transfer & Kiểu kiến trúc trao đổi trạng thái đại diện\\
  \textbf{SaaS} & Software as a Service & Phần mềm dưới dạng dịch vụ (thuê bao qua đám mây)\\
  \textbf{SHA} & Secure Hash Algorithm & Thuật toán băm an toàn\\
  \textbf{SMTP} & Simple Mail Transfer Protocol & Giao thức truyền thư điện tử đơn giản\\
  \textbf{SPA} & Single Page Application & Ứng dụng đơn trang\\
  \textbf{TMĐT} & & Thương mại điện tử\\
  \textbf{UI} & User Interface & Giao diện người dùng\\
  \textbf{URL} & Uniform Resource Locator & Định vị tài nguyên thống nhất (địa chỉ web)\\
  \textbf{UX} & User Experience & Trải nghiệm người dùng\\
  \textbf{VECOM} & Vietnam E-Commerce Association & Hiệp hội Thương mại điện tử Việt Nam\\

    \hline
\end{longtable}
